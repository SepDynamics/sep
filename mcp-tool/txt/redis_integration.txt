import { createClient } from 'redis';

class RedisIntegration {
    constructor(config) {
        this.config = config;
        this.client = null;
        this.connectionTimeout = 15000; // 15 second timeout
        this.maxRetries = 3;
        this.retryDelay = 1000;
        this.connectionPool = new Map();
        this.circuitBreaker = {
            failures: 0,
            lastFailure: 0,
            threshold: 5,
            resetTimeout: 30000
        };
        
        // Initialize fallback components
        this.fallbackComponents = {
            memoryTier: new Map(),
            coherenceTracker: new Map(),
            buildCache: new Map(),
            patternRegistry: new Map(),
            dependencyGraph: new Map()
        };
    }

    async initialize() {
        if (this.isCircuitBroken()) {
            console.log('Circuit breaker active, using fallback components');
            return false;
        }

        try {
            console.log('Redis integration initializing with config:', {
                enabled: this.config.enabled,
                host: this.config.host,
                port: this.config.port,
                socket: this.config.socket || process.env.SEP_REDIS_SOCKET,
                prefix: this.config.prefix,
                connectionTimeout: this.connectionTimeout,
                maxRetries: this.maxRetries
            });
            
            // Create Redis client with timeout
            this.client = createClient({
                socket: this.config.socket || process.env.SEP_REDIS_SOCKET ? {
                    path: this.config.socket || process.env.SEP_REDIS_SOCKET,
                    connectTimeout: this.connectionTimeout,
                } : {
                    host: this.config.host,
                    port: this.config.port,
                    connectTimeout: this.connectionTimeout,
                    reconnectStrategy: (retries) => {
                        if (retries >= this.maxRetries) {
                            console.error('Max Redis retries reached, failing over to fallback components');
                            this.recordFailure();
                            return new Error('Max retries reached');
                        }
                        return this.retryDelay;
                    }
                }
            });

            // Single error handler
            this.client.on('error', (err) => {
                console.error('Redis error:', {
                    code: err.code,
                    message: err.message,
                    command: err.command
                });
                this.recordFailure();
            });

            await Promise.race([
                this.client.connect(),
                new Promise((_, reject) =>
                    setTimeout(() => reject(new Error('Redis connection timeout after 15s')),
                    this.connectionTimeout)
                )
            ]);

            console.log('Redis connection established');
            return true;
        } catch (error) {
            console.warn('Failed to connect to Redis:', error);
            this.recordFailure();
            return false;
        }
    }

    isCircuitBroken() {
        if (this.circuitBreaker.failures >= this.circuitBreaker.threshold) {
            const timeSinceLastFailure = Date.now() - this.circuitBreaker.lastFailure;
            if (timeSinceLastFailure < this.circuitBreaker.resetTimeout) {
                return true;
            }
            // Reset circuit breaker after timeout
            this.circuitBreaker.failures = 0;
        }
        return false;
    }

    recordFailure() {
        this.circuitBreaker.failures++;
        this.circuitBreaker.lastFailure = Date.now();
    }

    isUsingRedis() {
        return this.client && this.client.isReady;
    }

    getComponent(name) {
        if (!this.isUsingRedis()) {
            console.log(`Using fallback component: ${name}`);
            return this.getFallbackComponent(name);
        }

        console.log(`Using Redis-backed component: ${name}`);
        return this.getRedisComponent(name);
    }

    getFallbackComponent(name) {
        const component = this.fallbackComponents[name];
        if (!component) {
            throw new Error(`Unknown fallback component: ${name}`);
        }
        return {
            get: async (key) => component.get(key),
            set: async (key, value) => component.set(key, value),
            delete: async (key) => component.delete(key),
            getAll: async () => Array.from(component.values()),
            clear: async () => component.clear()
        };
    }

    getRedisComponent(name) {
        const prefix = `${this.config.prefix || 'sep:'}${name}:`;
        
        switch (name) {
            case 'memoryTier':
                return {
                    storePattern: async (id, data) => {
                        const key = `${prefix}${id}`;
                        const fields = {
                            data: JSON.stringify(data.data || {}),
                            coherence: data.coherence.toString(),
                            contentType: data.contentType || '',
                            tier: data.tier || 'STM',
                            timestamp: new Date().toISOString()
                        };
                        await this.client.hSet(key, fields);
                    },
                    queryPatterns: async ({ tier, minCoherence, type }) => {
                        const keys = await this.client.keys(`${prefix}*`);
                        const patterns = [];
                        for (const key of keys) {
                            const pattern = await this.client.hGetAll(key);
                            if (pattern) {
                                const parsedPattern = {
                                    id: key.replace(prefix, ''),
                                    data: JSON.parse(pattern.data || '{}'),
                                    coherence: parseFloat(pattern.coherence),
                                    contentType: pattern.contentType,
                                    tier: pattern.tier,
                                    timestamp: pattern.timestamp
                                };
                                
                                if ((!tier || parsedPattern.tier === tier) &&
                                    (!minCoherence || parsedPattern.coherence >= minCoherence) &&
                                    (!type || parsedPattern.contentType === type)) {
                                    patterns.push(parsedPattern);
                                }
                            }
                        }
                        return patterns;
                    }
                };

            case 'coherenceTracker':
                return {
                    trackCoherence: async (id, score, metadata) => {
                        const key = `${prefix}${id}`;
                        const historyKey = `${key}:history`;
                        
                        // Store current state
                        const fields = {
                            score: score.toString(),
                            type: metadata.type || '',
                            deterministic: metadata.deterministic ? '1' : '0',
                            timestamp: metadata.timestamp || new Date().toISOString()
                        };
                        await this.client.hSet(key, fields);
                        
                        // Add to history list with generation tracking
                        const generation = await this.client.lLen(historyKey) + 1;
                        await this.client.lPush(historyKey, JSON.stringify({
                            generation,
                            coherence: score,
                            timestamp: fields.timestamp,
                            stability: metadata.deterministic ? 1.0 : 0.8
                        }));
                        
                        // Trim history to keep last N generations
                        await this.client.lTrim(historyKey, 0, 99); // Keep last 100 generations
                    },
                    getCoherenceHistory: async (id, generations = 10) => {
                        const key = `${prefix}${id}`;
                        const historyKey = `${key}:history`;
                        
                        // Get history entries
                        const entries = await this.client.lRange(historyKey, 0, generations - 1);
                        if (!entries || entries.length === 0) {
                            // Fallback to current state if no history
                            const current = await this.client.hGetAll(key);
                            if (!current || !current.score) {
                                return [];
                            }
                            return [{
                                generation: 1,
                                coherence: parseFloat(current.score),
                                timestamp: current.timestamp,
                                stability: current.deterministic === '1' ? 1.0 : 0.8
                            }];
                        }
                        
                        // Parse history entries
                        return entries.map(entry => JSON.parse(entry));
                    }
                };

            case 'buildCache':
                return {
                    get: async (key) => this.client.get(`${prefix}${key}`),
                    set: async (key, value) => this.client.set(`${prefix}${key}`, value, {
                        EX: 3600 // 1 hour expiry
                    })
                };

            case 'patternRegistry':
                return {
                    storePattern: async (id, pattern) => {
                        const key = `${prefix}${id}`;
                        await this.client.hSet(key, {
                            ...pattern,
                            timestamp: new Date().toISOString()
                        });
                    },
                    getAllPatterns: async () => {
                        const keys = await this.client.keys(`${prefix}*`);
                        const patterns = [];
                        for (const key of keys) {
                            const pattern = await this.client.hGetAll(key);
                            if (pattern) {
                                patterns.push(pattern);
                            }
                        }
                        return patterns;
                    }
                };

            case 'dependencyGraph':
                return {
                    addDependency: async (from, to, weight = 1.0) => {
                        const key = `${prefix}${from}`;
                        await this.client.zAdd(key, [{
                            score: weight,
                            value: to
                        }]);
                    }
                };

            default:
                throw new Error(`Unknown Redis component: ${name}`);
        }
    }

    async close() {
        if (this.client) {
            try {
                if (this.client.isReady) {
                    await this.client.quit();
                    console.log('Redis connection closed successfully');
                } else {
                    console.log('Redis client already disconnected');
                }
            } catch (error) {
                console.warn('Error while closing Redis connection:', error.message);
            } finally {
                this.client = null;
            }
        }
    }
}

export { RedisIntegration };
