/**
 * SEP MCP Controller
 * Central coordinator for SEP-MCP server integration with Redis Memory Tier
 */

import { RedisMemoryTier } from './redis_memory_tier.js';

/**
 * Base adapter class for MCP servers
 * Provides common interface and memory tier integration
 */
class BaseMCPAdapter {
    constructor(serverInstance, memoryTier) {
        this.server = serverInstance;
        this.memoryTier = memoryTier;
        this.serverName = serverInstance.serverConfig?.metadata?.name || 'unknown';
    }

    async processRequest(toolName, args) {
        try {
            // Get tool handler from server
            const tool = this.server.serverConfig.capabilities.tools.find(t => t.name === toolName);
            if (!tool) {
                throw new Error(`Tool ${toolName} not found in server ${this.serverName}`);
            }

            // Execute tool with args
            const result = await tool.handler(args);

            // Store result in memory tier if applicable
            if (result && typeof result === 'object') {
                const patternId = `${this.serverName}:${toolName}:${Date.now()}`;
                await this.memoryTier.storePattern(patternId, {
                    data: result,
                    coherence: result.coherenceScore || result.coherenceMetrics?.score || 0.5,
                    stability: result.stabilityScore || 0.5,
                    generations: 1,
                    tier: 'STM' // Start in STM, let memory tier handle promotion
                });
            }

            return result;
        } catch (error) {
            console.error(`Error processing request for ${this.serverName}:${toolName}:`, error);
            throw error;
        }
    }

    async validateRequest(toolName, args) {
        const tool = this.server.serverConfig.capabilities.tools.find(t => t.name === toolName);
        if (!tool) {
            return { valid: false, error: `Tool ${toolName} not found` };
        }

        // Validate against tool's input schema
        if (tool.inputSchema) {
            const { type, properties, required } = tool.inputSchema;
            
            // Check type
            if (type === 'object' && typeof args !== 'object') {
                return { valid: false, error: 'Arguments must be an object' };
            }

            // Check required properties
            if (required) {
                for (const prop of required) {
                    if (!(prop in args)) {
                        return { valid: false, error: `Missing required property: ${prop}` };
                    }
                }
            }

            // Validate property types if specified
            if (properties) {
                for (const [key, schema] of Object.entries(properties)) {
                    if (key in args) {
                        if (schema.type === 'string' && typeof args[key] !== 'string') {
                            return { valid: false, error: `Property ${key} must be a string` };
                        }
                        if (schema.enum && !schema.enum.includes(args[key])) {
                            return { valid: false, error: `Invalid value for ${key}. Must be one of: ${schema.enum.join(', ')}` };
                        }
                    }
                }
            }
        }

        return { valid: true };
    }
}

/**
 * Main SEP MCP Controller class
 * Handles server registration, request routing, and memory tier integration
 */
class SEPMCPController {
    constructor(options = {}) {
        this.serverRegistry = new Map();
        this.redisMemoryTier = null;
        this.initialized = false;
        this.options = {
            valkeyHost: options.valkeyHost || '127.0.0.1',
            valkeyPort: options.valkeyPort || 6379,
            valkeySocket: options.valkeySocket || '/run/valkey/sep-manifold.sock',
            valkeyPrefix: options.valkeyPrefix || 'sep:mcp:',
            coherenceThresholds: {
                minimum: options.coherenceThresholds?.minimum || 0.1,
                mtm_promotion: options.coherenceThresholds?.mtm_promotion || 0.7,
                ltm_promotion: options.coherenceThresholds?.ltm_promotion || 0.9,
                demotion: options.coherenceThresholds?.demotion || 0.3
            }
        };
    }

    async initialize() {
        if (this.initialized) {
            return;
        }

        try {
            // Initialize Redis Memory Tier
            this.redisMemoryTier = new RedisMemoryTier({
                host: this.options.valkeyHost,
                port: this.options.valkeyPort,
                socket: this.options.valkeySocket,
                prefix: this.options.valkeyPrefix,
                coherenceThresholds: this.options.coherenceThresholds
            });

            await this.redisMemoryTier.initialize();

            // Restore server registry from Redis
            try {
                const pattern = await this.redisMemoryTier.getPattern('server_registry');
                if (pattern && pattern.data) {
                    for (const [serverName, serverData] of Object.entries(pattern.data)) {
                        // Reconstruct server instance with all required methods
                        const serverInstance = {
                            serverConfig: {
                                ...serverData.instance.serverConfig,
                                capabilities: {
                                    ...serverData.instance.serverConfig.capabilities,
                                    tools: serverData.instance.serverConfig.capabilities.tools.map(tool => ({
                                        ...tool,
                                        handler: async (args) => {
                                            throw new Error('Handler not implemented');
                                        }
                                    }))
                                }
                            }
                        };

                        // Create adapter with proper prototype chain
                        const adapter = new BaseMCPAdapter(serverInstance, this.redisMemoryTier);
                        
                        this.serverRegistry.set(serverName, {
                            instance: serverInstance,
                            adapter
                        });
                    }
                    console.log(`Restored ${this.serverRegistry.size} servers from registry`);
                }
            } catch (restoreErr) {
                console.error('Failed to restore server registry:', restoreErr);
                // Continue initialization even if restore fails
            }
            
            this.initialized = true;
            console.log('SEP MCP Controller initialized successfully');
        } catch (error) {
            console.error('Failed to initialize SEP MCP Controller:', error);
            throw error;
        }
    }

    async registerServer(serverName, serverInstance, AdapterClass = BaseMCPAdapter) {
        if (!this.initialized) {
            await this.initialize();
        }

        if (this.serverRegistry.has(serverName)) {
            throw new Error(`Server ${serverName} is already registered`);
        }

        try {
            // Create adapter instance
            const adapter = new AdapterClass(serverInstance, this.redisMemoryTier);
            
            try {
                // Create server instance with required methods
                const serverWithMethods = {
                    ...serverInstance,
                    serverConfig: {
                        ...serverInstance.serverConfig,
                        capabilities: {
                            ...serverInstance.serverConfig.capabilities,
                            tools: serverInstance.serverConfig.capabilities.tools.map(tool => ({
                                ...tool,
                                handler: tool.handler || (async (args) => {
                                    throw new Error('Handler not implemented');
                                })
                            }))
                        }
                    }
                };

                // Create adapter with proper prototype chain
                const adapterInstance = new AdapterClass(serverWithMethods, this.redisMemoryTier);

                // Store in registry
                this.serverRegistry.set(serverName, {
                    instance: serverWithMethods,
                    adapter: adapterInstance
                });

                // Store server metadata and registry in Redis
                const metadata = {
                    name: serverName,
                    capabilities: serverWithMethods.serverConfig.capabilities,
                    registered_at: new Date().toISOString()
                };

                // Store individual server metadata
                await this.redisMemoryTier.storePattern(`server:${serverName}`, {
                    data: metadata,
                    coherence: 1.0,
                    stability: 1.0,
                    generations: 1,
                    tier: 'LTM' // Server metadata should be in LTM
                });

                // Update and store full registry
                const registryData = {};
                for (const [name, data] of this.serverRegistry.entries()) {
                    const serverTools = data.instance.serverConfig.capabilities.tools.map(tool => ({
                        name: tool.name,
                        description: tool.description,
                        inputSchema: tool.inputSchema,
                        handler: tool.handler ? true : false // Just store if handler exists
                    }));

                    registryData[name] = {
                        instance: {
                            serverConfig: {
                                ...data.instance.serverConfig,
                                capabilities: {
                                    ...data.instance.serverConfig.capabilities,
                                    tools: serverTools
                                }
                            }
                        }
                    };
                }

                await this.redisMemoryTier.storePattern('server_registry', {
                    data: registryData,
                    coherence: 1.0,
                    stability: 1.0,
                    generations: 1,
                    tier: 'LTM'
                });

                console.log(`Server ${serverName} registered successfully`);
                return adapterInstance;
            } catch (error) {
                console.error(`Error during server registration:`, error);
                // Clean up any partial registration
                this.serverRegistry.delete(serverName);
                await this.redisMemoryTier.deletePattern(`server:${serverName}`);
                throw error;
            }

            console.log(`Server ${serverName} registered successfully`);
            return adapter;
        } catch (error) {
            console.error(`Failed to register server ${serverName}:`, error);
            throw error;
        }
    }

    async routeRequest(serverName, toolName, args) {
        if (!this.initialized) {
            await this.initialize();
        }

        const server = this.serverRegistry.get(serverName);
        if (!server) {
            throw new Error(`Server ${serverName} not found`);
        }

        // Validate request
        const validation = await server.adapter.validateRequest(toolName, args);
        if (!validation.valid) {
            throw new Error(`Invalid request: ${validation.error}`);
        }

        // Process request through adapter
        return await server.adapter.processRequest(toolName, args);
    }

    async getServerCapabilities(serverName) {
        const server = this.serverRegistry.get(serverName);
        if (!server) {
            throw new Error(`Server ${serverName} not found`);
        }
        return server.instance.serverConfig.capabilities;
    }

    async getAllServers() {
        return Array.from(this.serverRegistry.keys());
    }

    async deregisterServer(serverName) {
        if (!this.initialized) {
            await this.initialize();
        }

        if (!this.serverRegistry.has(serverName)) {
            throw new Error(`Server ${serverName} is not registered`);
        }

        try {
            // Remove from registry
            this.serverRegistry.delete(serverName);

            // Remove server metadata from Redis
            await this.redisMemoryTier.deletePattern(`server:${serverName}`);

            // Update full registry
            const registryData = {};
            for (const [name, data] of this.serverRegistry.entries()) {
                registryData[name] = {
                    instance: {
                        serverConfig: data.instance.serverConfig
                    },
                    adapter: {
                        serverName: data.adapter.serverName
                    }
                };
            }

            await this.redisMemoryTier.storePattern('server_registry', {
                data: registryData,
                coherence: 1.0,
                stability: 1.0,
                generations: 1,
                tier: 'LTM'
            });

            console.log(`Server ${serverName} deregistered successfully`);
        } catch (error) {
            console.error(`Failed to deregister server ${serverName}:`, error);
            throw error;
        }
    }

    async close() {
        if (this.redisMemoryTier) {
            await this.redisMemoryTier.close();
        }
        this.initialized = false;
        this.serverRegistry.clear();
        console.log('SEP MCP Controller shut down successfully');
    }
}

export { SEPMCPController, BaseMCPAdapter };
